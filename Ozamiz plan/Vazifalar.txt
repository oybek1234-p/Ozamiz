fun getDailyTasks(userPoints: Int, streak: Int): List<Task> {
    val difficulty = when {
        userPoints < 500 -> 1..3  // Oson
        userPoints in 500..2000 -> 4..7 // O'rta
        else -> 8..10 // Qiyin
    }
    // Bazadan ushbu darajadagi 3 ta random vazifani qaytaradi
    return database.getTasksByDifficulty(difficulty).random(3)
}
